% ---------- Research interests ----------
\section{Research Interests}
X-ray reflection from accreting black holes, in both X-ray binaries and AGN.
I study how the geometry of the corona shapes the illumination of the accretion
disc and, in turn, the observed reflection spectrum, motivated by recent
IXPE polarimetry constraints. I combine general-relativistic ray tracing with
X-ray spectroscopy, and aim to develop a general method applicable to any
axisymmetric coronal geometry across a broad range of sources.

\section{Research Experience}
\entry{2025 -- present}{%
  \textbf{Coronal geometry and disc illumination in MAXI J1348$-$630}
  \begin{itemize}
    \item Modelling disc illumination profiles for lamppost, ring and disc coronae,
          building towards a general method for arbitrary axisymmetric geometries.
    \item Fitting the resulting reflection spectra to NuSTAR observations of the black hole
          X-ray binary MAXI J1348$-$630, using a NuSTAR reduction pipeline I developed.
    \item GR ray tracing with Gradus.jl and Kerrz (Julia); HPC on Isambard 3 / Isambard-AI.
    \item Collaborators: Dr Jiachen Jiang (Warwick), Dr Fergus Baker (Newcastle).
          Paper in preparation.
  \end{itemize}}
 
\entry{2024 -- 2025}{%
  \textbf{Relativistic absorption as an alternative to ultra-fast outflows in AGN}
  \begin{itemize}
    \item Tested whether broad absorption features usually attributed to ultra-fast outflows
          can instead arise from relativistically broadened absorption by material
          co-rotating close to the black hole.
    \item Archival XMM-Newton data of IRAS 13224$-$3809, using an XMM-Newton reduction pipeline I developed.
  \end{itemize}}
